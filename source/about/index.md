<script>
// 关于页：一屏锁定 + 横向翻页册 + 终端打字机。四路输入（箭头/圆点、键盘、
// 滚轮、触摸）全部接到同一个 go()，翻页状态只在这里改。无 jQuery。
(function () {
  var page = document.querySelector('.about-page');
  if (!page) return;
  var deck = page.querySelector('.about-deck');
  var slides = Array.prototype.slice.call(page.querySelectorAll('.about-slide'));
  var dots = Array.prototype.slice.call(page.querySelectorAll('.about-dot'));
  var arrows = Array.prototype.slice.call(page.querySelectorAll('.about-arrow'));
  var count = page.querySelector('.about-count');
  if (!deck || slides.length < 2) return;

  var total = slides.length;
  var cur = 0;

  function pad(n) { return (n < 10 ? '0' : '') + n; }

  /* ---- 终端打字机 ------------------------------------------------------
     输出行默认是可见的；只有 JS 真的跑起来（给 .about-page 挂上 term-js）
     才会被 CSS 藏起来等待逐条点亮。这样「脚本挂了」和「禁用 JS」这两种
     最糟的情况退化成「内容全都在、只是没有动画」，而不是一片空白。

     命令原文在这里一次性抓走：打字过程中会改写 textContent，边打边读会
     读到半截字符串。 */
  var reduce = !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  var cmds = [];
  Array.prototype.forEach.call(page.querySelectorAll('.term-cmd[data-type]'), function (el) {
    cmds.push({ el: el, text: el.textContent });
  });
  page.classList.add('term-js');

  function finish(slide) {
    slide.classList.remove('is-typing');
    slide.classList.add('is-typed');
    Array.prototype.forEach.call(slide.querySelectorAll('.term-out'), function (el) {
      el.classList.add('is-on');
    });
  }

  function typeSlide(slide) {
    if (slide.getAttribute('data-term-done')) return;
    slide.setAttribute('data-term-done', '1');
    if (reduce) { finish(slide); return; }

    slide.classList.add('is-typing');
    // 兜底闸：无论中间出什么岔子（DOM 变了、定时器被杀），到点必定把这一屏
    // 的内容全部放出来。宁可没有动画，也不能把内容留在 visibility:hidden。
    var guard = setTimeout(function () { finish(slide); }, 9000);
    try {
      var screen = slide.querySelector('.term-screen');
      if (!screen) { clearTimeout(guard); finish(slide); return; }

      // 按块跑：一条命令行 + 紧随其后的输出，才像真的在敲终端
      var blocks = [];
      Array.prototype.forEach.call(screen.children, function (el) {
        var cmd = el.querySelector ? el.querySelector('.term-cmd[data-type]') : null;
        if (cmd) blocks.push({ cmd: cmd, outs: [] });
        else if (blocks.length) blocks[blocks.length - 1].outs.push(el);
      });

      var bi = 0;
      function nextBlock() {
        if (bi >= blocks.length) { clearTimeout(guard); finish(slide); return; }
        var b = blocks[bi++];
        var full = b.cmd.textContent;
        for (var i = 0; i < cmds.length; i++) {
          if (cmds[i].el === b.cmd) { full = cmds[i].text; break; }
        }
        b.cmd.textContent = '';
        var ci = 0;
        (function tick() {
          if (ci >= full.length) {
            b.outs.forEach(function (el, k) {
              el.style.setProperty('--d', String(k));
              el.classList.add('is-on');
            });
            setTimeout(nextBlock, 70);
            return;
          }
          b.cmd.textContent = full.slice(0, ++ci);
          setTimeout(tick, 18);
        })();
      }
      nextBlock();
    } catch (e) {
      clearTimeout(guard);
      finish(slide);
    }
  }

  function go(n, force) {
    n = Math.max(0, Math.min(total - 1, n));
    if (n === cur && !force) return;

    var prev = slides[cur];
    var next = slides[n];
    // 焦点在即将退场的那一屏里时要先搬走，见下面的顺序说明
    var focusWasInside = prev !== next && prev.contains(document.activeElement);

    cur = n;

    page.style.setProperty('--about-i', String(n));
    page.style.setProperty('--about-p', String((n + 1) / total));

    // 顺序不能反：先把新屏点亮（is-active 让它从 visibility:hidden 变可见，
    // 否则 focus() 落不上去），搬完焦点，最后才给其余屏挂 inert。
    // 直接给含焦点的祖先加 inert 会把焦点弹到 body，键盘用户失去位置；
    // 对含焦点的祖先加 aria-hidden 本身也是 ARIA 违规。
    // prev !== next 的判断不能省：首次 go() 时两者是同一屏（cur 与目标都是
    // 第 1 屏），加了又删会互相抵消，把服务端写死的首屏点亮抹掉 → 整页空白。
    next.classList.add('is-active');
    if (prev !== next) prev.classList.remove('is-active');
    next.removeAttribute('inert');
    next.removeAttribute('aria-hidden');

    if (focusWasInside) {
      if (!next.hasAttribute('tabindex')) next.setAttribute('tabindex', '-1');
      try { next.focus({ preventScroll: true }); } catch (e) { next.focus(); }
    }

    slides.forEach(function (s) {
      if (s === next) return;
      s.setAttribute('inert', '');
      s.setAttribute('aria-hidden', 'true');
    });

    dots.forEach(function (d, i) {
      if (i === n) d.setAttribute('aria-current', 'true');
      else d.removeAttribute('aria-current');
    });

    if (count) {
      count.innerHTML = '<span aria-hidden="true">' + pad(n + 1) + ' / ' + pad(total) + '</span>' +
        '<span class="sr-only">第 ' + (n + 1) + ' 屏，共 ' + total + ' 屏</span>';
    }

    arrows.forEach(function (a) {
      var dir = a.getAttribute('data-dir');
      a.disabled = (dir === 'prev' && n === 0) || (dir === 'next' && n === total - 1);
    });

    // 打字机跟着翻页走：切到哪屏才开始敲哪屏，已经敲过的不重来
    typeSlide(next);

    // #p{n} 深链：刷新/收藏/分享能回到同一屏。replaceState 不触发滚动
    // （页面上没有 id="p{n}" 的元素）。第 1 屏只在 hash 本来就是 #p{n}
    // 时才清掉，不动别人留下的 hash。
    if (history.replaceState) {
      if (n === 0) {
        if (/^#p\d+$/.test(location.hash || '')) {
          history.replaceState(null, '', location.pathname + location.search);
        }
      } else {
        history.replaceState(null, '', '#p' + (n + 1));
      }
    }
  }

  dots.forEach(function (d, i) {
    d.addEventListener('click', function () { go(i); });
  });
  arrows.forEach(function (a) {
    a.addEventListener('click', function () {
      go(cur + (a.getAttribute('data-dir') === 'next' ? 1 : -1));
    });
  });

  // 键盘。输入态与已被别人处理过的事件一律让开：站内 after-footer.ejs
  // 已经用 window 级 keydown 处理 Ctrl+K 与 "/"，不能抢。
  function isTyping(el) {
    if (!el || !el.tagName) return false;
    var t = el.tagName;
    return t === 'INPUT' || t === 'TEXTAREA' || t === 'SELECT' || el.isContentEditable === true;
  }
  document.addEventListener('keydown', function (e) {
    if (e.defaultPrevented || e.ctrlKey || e.metaKey || e.altKey) return;
    if (isTyping(e.target)) return;
    var k = e.key;
    if (k === 'ArrowRight' || k === 'PageDown') { e.preventDefault(); go(cur + 1); }
    else if (k === 'ArrowLeft' || k === 'PageUp') { e.preventDefault(); go(cur - 1); }
    else if (k === 'Home') { e.preventDefault(); go(0); }
    else if (k === 'End') { e.preventDefault(); go(total - 1); }
  });

  // 滚轮。三个坑都在这里：
  // 1) Ctrl/⌘+滚轮（含 macOS 触控板捏合）走的是 wheel 且 ctrlKey=true，
  //    无条件 preventDefault 会让用户没法缩放页面 —— 直接放行。
  // 2) deltaMode 必须归一化：Firefox 常以「行」派发，deltaY 只有 3 上下，
  //    拿像素阈值去比永远触发不了，表现为 Firefox 上滚轮完全不翻页。
  // 3) 触控板的惯性尾流会持续一两秒，定时器节流挡不住 —— 改成「过阈值立刻
  //    翻页并上锁，滚动静止 140ms 后才解锁」，惯性期间不断重置解锁计时器。
  var acc = 0;
  var locked = false;
  var quiet = null;
  var THRESH = 40;

  function deltaPx(e) {
    var m = e.deltaMode === 1 ? 16 : (e.deltaMode === 2 ? 100 : 1);
    return e.deltaY * m;
  }
  // 内层还能滚就交给它，不拦
  function canScroll(el, dy) {
    if (!el) return false;
    var max = el.scrollHeight - el.clientHeight;
    if (max <= 1) return false;
    if (dy < 0) return el.scrollTop > 0;
    return el.scrollTop < max - 1;
  }

  page.addEventListener('wheel', function (e) {
    if (e.ctrlKey || e.metaKey) return;
    var inner = e.target && e.target.closest ? e.target.closest('.about-slide-body') : null;
    if (canScroll(inner, e.deltaY)) return;
    e.preventDefault();

    clearTimeout(quiet);
    quiet = setTimeout(function () { acc = 0; locked = false; }, 140);
    if (locked) return;

    acc += deltaPx(e);
    if (Math.abs(acc) >= THRESH) {
      var d = acc > 0 ? 1 : -1;
      acc = 0;
      locked = true;
      go(cur + d);
    }
  }, { passive: false });

  // 触摸：水平位移够大且明显压过纵向才算翻页（同 source/lyrics/index.md 的
  // 既有判定）。全程不 preventDefault，纵向留给屏内自己滚。
  var tX = 0;
  var tY = 0;
  var tOk = false;
  page.addEventListener('touchstart', function (e) {
    if (e.touches.length !== 1) { tOk = false; return; }
    tOk = true;
    tX = e.touches[0].clientX;
    tY = e.touches[0].clientY;
  }, { passive: true });
  page.addEventListener('touchend', function (e) {
    if (!tOk) return;
    tOk = false;
    var t = e.changedTouches && e.changedTouches[0];
    if (!t) return;
    var dx = t.clientX - tX;
    var dy = t.clientY - tY;
    if (Math.abs(dx) < 50 || Math.abs(dx) < Math.abs(dy) * 1.5) return;
    go(cur + (dx < 0 ? 1 : -1));
  }, { passive: true });

  /* ---- 构建期数据 ------------------------------------------------------
     /about-feed.json 由 scripts/about-feed.js 在构建时生成。取不到就沉默
     保留 HTML 里写死的兜底数字（那组数字在构建那一刻是对的），不报错、
     不清空 —— 统计区退化成「略微过时」而不是「空着」。 */
  function applyFeed(d) {
    var nums = ['posts', 'categories', 'tags', 'since', 'updated'];
    nums.forEach(function (k) {
      var el = page.querySelector('[data-feed="' + k + '"]');
      if (el && d[k] !== undefined && d[k] !== null) el.textContent = String(d[k]);
    });

    var tagEl = page.querySelector('p[data-feed="tags"]');
    if (tagEl && d.topTags && d.topTags.length) {
      tagEl.textContent = d.topTags.map(function (t) {
        return t.name + '(' + t.count + ')';
      }).join('  ');
    }

    var list = page.querySelector('ul[data-feed="recent"]');
    if (list && d.recent && d.recent.length) {
      list.textContent = '';
      d.recent.forEach(function (p) {
        var li = document.createElement('li');
        var d0 = document.createElement('span');
        d0.className = 'term-key';
        d0.textContent = String(p.date || '').slice(5);
        var a = document.createElement('a');
        a.href = p.url;
        a.textContent = p.title;
        li.appendChild(d0);
        li.appendChild(a);
        list.appendChild(li);
      });
    }
  }

  if (window.fetch) {
    fetch('/about-feed.json', { credentials: 'same-origin' })
      .then(function (r) { return r.ok ? r.json() : null; })
      .then(function (d) { if (d) applyFeed(d); })
      .catch(function () { /* 保留兜底数字 */ });
  }

  // 初始屏：第 1 屏的 is-active 已服务端写死在 HTML 里（否则脚本执行前 5 片
  // 全 opacity:0，是必现的空白闪）；这里只处理 #p{n} 深链与首屏状态同步。
  var m = /^#p(\d+)$/.exec(location.hash || '');
  go(m ? parseInt(m[1], 10) - 1 : 0, true);
})();
</script>
